cmake_minimum_required(VERSION 3.28)
project(CrosshairNative VERSION 0.1.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_CXX_SCAN_FOR_MODULES OFF)
enable_testing()

if(MSVC)
  add_compile_options(/W4 /permissive- /utf-8 /EHsc)
  add_compile_definitions(UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
endif()
add_library(crosshair_core core/model.cpp)
target_include_directories(crosshair_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
add_executable(core_tests tests/unit/core_tests.cpp)
target_link_libraries(core_tests PRIVATE crosshair_core)
add_test(NAME core COMMAND core_tests)

if(WIN32)
  enable_language(RC)
  if(NOT MSVC)
    message(FATAL_ERROR "The Windows engine requires MSVC and the Windows SDK.")
  endif()
  add_library(crosshair_rendering rendering/renderer.cpp)
  target_link_libraries(crosshair_rendering PUBLIC crosshair_core d2d1 windowscodecs ole32)
  add_library(crosshair_storage apps/engine/storage.cpp)
  target_link_libraries(crosshair_storage PUBLIC crosshair_core windowsapp shell32)
  target_link_libraries(crosshair_rendering PUBLIC crosshair_storage)
  add_library(crosshair_presets core/preset_json.cpp)
  target_link_libraries(crosshair_presets PUBLIC crosshair_storage)
  add_library(crosshair_pack core/pack.cpp)
  target_link_libraries(crosshair_pack PUBLIC crosshair_presets crypt32)
  add_executable(catalog_builder tools/catalog-builder/main.cpp)
  target_link_libraries(catalog_builder PRIVATE crosshair_presets crosshair_rendering)
  add_test(NAME catalog COMMAND catalog_builder "${CMAKE_CURRENT_BINARY_DIR}/catalog")
  add_library(crosshair_ipc apps/ipc.cpp)
  target_link_libraries(crosshair_ipc PUBLIC crosshair_core advapi32)
  add_executable(CrosshairNative.Engine WIN32 apps/engine/main.cpp apps/engine/app.manifest apps/assets/icon.rc)
  target_include_directories(CrosshairNative.Engine PRIVATE apps/assets)
  target_link_libraries(CrosshairNative.Engine PRIVATE crosshair_rendering crosshair_storage crosshair_presets crosshair_ipc
    user32 gdi32 shell32 advapi32 dwmapi wtsapi32)
  add_executable(CrosshairNative.TestTarget WIN32 tools/test-target/main.cpp apps/engine/app.manifest)
  target_link_libraries(CrosshairNative.TestTarget PRIVATE crosshair_core user32 gdi32)
  add_executable(renderer_tests tests/rendering/renderer_tests.cpp)
  target_link_libraries(renderer_tests PRIVATE crosshair_rendering)
  add_test(NAME renderer COMMAND renderer_tests)
  add_executable(storage_tests tests/integration/storage_tests.cpp)
  target_link_libraries(storage_tests PRIVATE crosshair_storage crosshair_presets)
  add_test(NAME storage COMMAND storage_tests)
  add_executable(ipc_tests tests/integration/ipc_tests.cpp)
  target_link_libraries(ipc_tests PRIVATE crosshair_ipc user32)
  add_test(NAME ipc COMMAND ipc_tests)
  add_executable(pack_tests tests/integration/pack_tests.cpp)
  target_link_libraries(pack_tests PRIVATE crosshair_pack crosshair_rendering)
  add_test(NAME pack COMMAND pack_tests)
  add_executable(engine_tests tests/integration/engine_tests.cpp)
  target_link_libraries(engine_tests PRIVATE crosshair_ipc crosshair_presets ole32)
  add_test(NAME engine COMMAND engine_tests)
  set_tests_properties(engine PROPERTIES TIMEOUT 30)
endif()
